# 1. Open Ubuntu using WSL
wsl -d Ubuntu


# 2. Update package lists
sudo apt update


# 3. Install Ansible
sudo apt install ansible -y


# 4. Verify Ansible installation
ansible --version


# 5. Create a directory
mkdir ansible_lab


# 6. Navigate to the directory
cd ansible_lab


# 7. Create a playbook file
nano basic_playbook.yml


# 8. Execute the Ansible playbook
ansible-playbook basic_playbook.yml


# 9. List files in the home directory
ls /home/zoya


# 10. Open the created text file
nano /home/zoya/ansible_file.txt


# 11. Display the message file content
cat /home/zoya/ansible_message.txt
